local urls = {
	"http://127.0.0.1:8787/script.lua",
	"http://127.0.0.1:8787/cryax.lua",
}

local function looks_like_script(src)
	if type(src) ~= "string" or #src < 80 then
		return false
	end
	local head = string.gsub(string.sub(src, 1, 200), "^%s+", "")
	if string.sub(head, 1, 1) == "<" then
		return false
	end
	if string.find(string.sub(head, 1, 40), "404:", 1, true) then
		return false
	end
	return true
end

local function download(url)
	local src
	pcall(function()
		src = game:HttpGet(url)
	end)
	if not looks_like_script(src) then
		pcall(function()
			src = game:HttpGet(url, true)
		end)
	end
	if not looks_like_script(src) then
		local req = http_request or request or (syn and syn.request)
		if type(req) == "function" then
			local ok, res = pcall(req, { Url = url, Method = "GET" })
			if ok and type(res) == "table" then
				src = res.Body or res.body
			elseif ok and type(res) == "string" then
				src = res
			end
		end
	end
	if looks_like_script(src) then
		return src
	end
	return nil
end

local src
for i = 1, #urls do
	src = download(urls[i])
	if src then
		break
	end
end

if not src then
	error("Cryax download failed. Start the host first:  cd %USERPROFILE%\\cryax\\host ; .\\start.ps1 -Hidden")
end

local compile = loadstring or load
if type(compile) ~= "function" then
	error("Cryax: this executor has no loadstring/load")
end
local fn, err = compile(src)
if not fn then
	error("Cryax loadstring failed: " .. tostring(err))
end
fn()
